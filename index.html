/* ===================================================================== Maytoni Group · Приём ответов анкеты перед обучением Google Apps Script — вставить в Extensions → Apps Script новой Google-таблицы, развернуть как веб-приложение и вписать полученный URL в SUBMIT_URL внутри файла анкеты (Анкета_перед_обучением_Maytoni.html). Что делает: 1. doPost(e) — принимает JSON от анкеты, дописывает строку в лист «Ответы» (создаёт его и шапку при первом запуске). 2. Для каждого выбранного участником направления считает и пишет КЛЮЧ УРОВНЯ — 🔴 Новичок / 🔴 Приоритет / 🟡 Рабочий / 🟢 Уверенный — в отдельный столбец "<Направление>LevelKey". Так по цвету эмодзи в таблице сразу видно, кто новичок, а кто уже уверенно работает, без открытия каждой анкеты по отдельности. 3. После каждого ответа пересобирает лист «Сводка» — по каждому направлению: сколько всего отметили, сколько в каждом уровне, средняя самооценка (Confidence) и готовая рекомендация тренеру. 4. Условное форматирование (цветная заливка ячеек LevelKey в листе «Ответы») настраивается один раз командой меню «Анкета КУ → Настроить форматирование» (или автоматически при первом ответе). Ничего в этом файле менять не нужно, если состав направлений/вопросов анкеты не менялся с текущей версии — DIRECTIONS_SCHEMA ниже зеркалит структуру direction-модулей из анкеты 1:1. ===================================================================== */ const SHEET_RESPONSES = "Ответы"; const SHEET_SUMMARY = "Сводка"; /* Схема направлений — та же, что DIRECTION_ORDER/DIRECTIONS в анкете. type: "scored" — направления через makeSimpleDirection (outdoor, warmfloor, eui, led, control): Level/Confidence/Novice + 3 ситуации с баллами 0–2 и полями Solution/Product/Commerce/TotalScore. "decor" — Интерьерный свет: свой формат, 6 ситуаций, 3 отдельных балла (Ustanovka/Commerce/Navyk), без TotalScore. "plain" — Архитектурный свет: без баллов вообще, чек-бокс "какие ситуации знакомы" + отдельная шкала уверенности против конкурента (archCompetitorConfidence). */ const DIRECTIONS_SCHEMA = [ { key: "arch", label: "Архитектурный свет", type: "plain" }, { key: "decor", label: "Интерьерный свет", type: "decor" }, { key: "outdoor", label: "Уличное освещение", type: "scored" }, { key: "warmfloor", label: "Тёплый пол", type: "scored" }, { key: "eui", label: "Электроустановки", type: "scored" }, { key: "led", label: "Светодиодная лента", type: "scored" }, { key: "control", label: "Управление освещением", type: "scored" } ]; const BASE_COLUMNS = ["submittedAt", "eventId", "stage", "role", "experience", "city", "partner", "directions", "preferredFormats", "trainerQuestion"]; function columnsForDirection(d) { if (d.type === "scored") { const p = d.key; return [ p + "Level", p + "Confidence", p + "NoviceDifficulties", p + "Situation1", p + "Situation2", p + "Situation3", p + "SolutionScore", p + "ProductScore", p + "CommerceScore", p + "TotalScore", p + "Profile", p + "SelfAssessment", p + "LevelKey" ]; } if (d.type === "decor") { return [ "decorLevel", "decorConfidence", "decorNoviceDifficulties", "decorUstanovka", "decorCommerce", "decorNavyk1", "decorNavyk2", "decorUstanovkaScore", "decorCommerceScore", "decorNavykScore", "decorProfile", "decorFocus", "decorSelfAssessment", "decorLevelKey" ]; } // plain — arch return [ "archLevel", "archConfidence", "archNoviceDifficulties", "archWeakAreas", "archCompetitorConfidence", "archLevelKey" ]; } function allColumns() { let cols = BASE_COLUMNS.slice(); DIRECTIONS_SCHEMA.forEach(d => { cols = cols.concat(columnsForDirection(d)); }); return cols; } /* --------------------- КЛЮЧ УРОВНЯ (визуальная оценка) ---------------- Единая шкала независимо от того, есть у направления баллы или нет: 🔴 Новичок — участник отметил уровень "почти не работаю" 🔴 Приоритет — работает с направлением, но результат слабый 🟡 Рабочий — средний, есть точечные зоны роста 🟢 Уверенный — сильный результат, можно усложнять программу Пусто — участник это направление не выбирал. */ function levelKey(prefix, data) { const level = data[prefix + "Level"]; if (!level) return ""; if (level === "novice") return "🔴 Новичок"; const total = Number(data[prefix + "TotalScore"]); if (!isNaN(total) && data[prefix + "TotalScore"] !== "" && data[prefix + "TotalScore"] !== undefined) { if (total >= 5) return "🟢 Уверенный"; if (total >= 3) return "🟡 Рабочий"; return "🔴 Приоритет"; } if (prefix === "decor") { // v2: все три измерения по шкале 0–2, максимум суммы теперь 6 (как у "scored") const sum = (Number(data.decorUstanovkaScore) || 0) + (Number(data.decorCommerceScore) || 0) + (Number(data.decorNavykScore) || 0); if (sum >= 5) return "🟢 Уверенный"; if (sum >= 3) return "🟡 Рабочий"; return "🔴 Приоритет"; } if (prefix === "arch") { // без баллов — используем шкалу уверенности против конкурента (1–10) const c = Number(data.archCompetitorConfidence) || 0; if (c >= 8) return "🟢 Уверенный"; if (c >= 4) return "🟡 Рабочий"; return "🔴 Приоритет"; } return ""; } /* ---------------------------- Приём ответа ----------------------------- */ function doPost(e) { try { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); const cols = allColumns(); let sheet = ss.getSheetByName(SHEET_RESPONSES); let firstRun = false; if (!sheet) { sheet = ss.insertSheet(SHEET_RESPONSES); sheet.appendRow(cols); sheet.setFrozenRows(1); firstRun = true; } DIRECTIONS_SCHEMA.forEach(d => { data[d.key + "LevelKey"] = levelKey(d.key, data); }); const row = cols.map(c => { const v = data[c]; if (Array.isArray(v)) return v.join("; "); return (v === undefined || v === null) ? "" : v; }); sheet.appendRow(row); if (firstRun) applyConditionalFormatting(sheet, cols); updateSummary(ss, cols); return ContentService.createTextOutput(JSON.stringify({ status: "ok" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(err) })) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { return ContentService.createTextOutput(JSON.stringify({ status: "ok", message: "Анкета КУ — приём ответов работает" })) .setMimeType(ContentService.MimeType.JSON); } /* ------------------- Условное форматирование листа «Ответы» ------------ */ function applyConditionalFormatting(sheet, cols) { const rules = []; DIRECTIONS_SCHEMA.forEach(d => { const colName = d.key + "LevelKey"; const colIndex = cols.indexOf(colName) + 1; // 1-based if (colIndex <= 0) return; const range = sheet.getRange(2, colIndex, Math.max(sheet.getMaxRows() - 1, 1), 1); rules.push(SpreadsheetApp.newConditionalFormatRule() .whenTextContains("🟢").setBackground("#d9ead3").setRanges([range]).build()); rules.push(SpreadsheetApp.newConditionalFormatRule() .whenTextContains("🟡").setBackground("#fff2cc").setRanges([range]).build()); rules.push(SpreadsheetApp.newConditionalFormatRule() .whenTextContains("🔴").setBackground("#f4cccc").setRanges([range]).build()); }); sheet.setConditionalFormatRules(rules); } /* ----------------------------- Лист «Сводка» ---------------------------- */ /* Пересобирается заново при каждом ответе — всегда актуальная картина по группе. Специально не через формулы Sheets, а прямым подсчётом в скрипте, чтобы не зависеть от точных номеров столбцов при правках. */ function updateSummary(ss, cols) { const respSheet = ss.getSheetByName(SHEET_RESPONSES); const lastRow = respSheet.getLastRow(); let summary = ss.getSheetByName(SHEET_SUMMARY); if (!summary) summary = ss.insertSheet(SHEET_SUMMARY); summary.clear(); const header = ["Направление", "Отметили", "🔴 Новичок", "🔴 Приоритет", "🟡 Рабочий", "🟢 Уверенный", "Ср. уверенность (1–10)", "Рекомендация"]; summary.appendRow(header); summary.setFrozenRows(1); summary.getRange(1, 1, 1, header.length).setFontWeight("bold").setBackground("#efefef"); if (lastRow < 2) { summary.autoResizeColumns(1, header.length); return; } const values = respSheet.getRange(2, 1, lastRow - 1, cols.length).getValues(); DIRECTIONS_SCHEMA.forEach(d => { const levelIdx = cols.indexOf(d.key + "Level"); const keyIdx = cols.indexOf(d.key + "LevelKey"); const confIdx = cols.indexOf(d.key + "Confidence"); let total = 0, novice = 0, priority = 0, working = 0, confident = 0; let confSum = 0, confCount = 0; values.forEach(row => { const level = row[levelIdx]; if (!level) return; total++; const key = row[keyIdx]; if (key === "🔴 Новичок") novice++; else if (key === "🔴 Приоритет") priority++; else if (key === "🟡 Рабочий") working++; else if (key === "🟢 Уверенный") confident++; const conf = Number(row[confIdx]); if (!isNaN(conf) && row[confIdx] !== "") { confSum += conf; confCount++; } }); const avgConf = confCount ? (confSum / confCount).toFixed(1) : ""; let recommendation = "—"; if (total > 0) { const redShare = (novice + priority) / total; const greenShare = confident / total; if (redShare >= 0.5) recommendation = "⚠️ Нужен базовый блок — больше половины группы новички или слабая практика"; else if (greenShare >= 0.5) recommendation = "✅ Можно усложнять — больше половины группы уверенно работает"; else recommendation = "Смешанная группа — предусмотреть разноуровневые задания"; } summary.appendRow([d.label, total, novice, priority, working, confident, avgConf, recommendation]); }); summary.autoResizeColumns(1, header.length); } /* ------------------------------- Меню ----------------------------------- */ function onOpen() { SpreadsheetApp.getUi() .createMenu("Анкета КУ") .addItem("Настроить форматирование", "manualSetupFormatting") .addItem("Обновить сводку", "manualUpdateSummary") .addToUi(); } function manualSetupFormatting() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_RESPONSES); if (!sheet) { SpreadsheetApp.getUi().alert("Лист «Ответы» ещё не создан — он появится после первого ответа анкеты."); return; } applyConditionalFormatting(sheet, allColumns()); SpreadsheetApp.getUi().alert("Готово: условное форматирование настроено."); } function manualUpdateSummary() { const ss = SpreadsheetApp.getActiveSpreadsheet(); if (!ss.getSheetByName(SHEET_RESPONSES)) { SpreadsheetApp.getUi().alert("Лист «Ответы» ещё не создан — он появится после первого ответа анкеты."); return; } updateSummary(ss, allColumns()); SpreadsheetApp.getUi().alert("Готово: сводка обновлена."); }
